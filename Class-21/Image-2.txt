import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(

      home: Scaffold(

        body: SingleChildScrollView(

                height: 150,

              const SizedBox(height: 20),

              const Text("BoxFit.contain"),
              Container(
                width: 300,
                height: 150,
                color: Colors.grey,
                child: Image.asset(
                  'assets/images/student.jpg',
                  fit: BoxFit.contain,
                ),
              ),

              const SizedBox(height: 20),

              const Text("BoxFit.fill"),
              Container(
                width: 300,
                height: 150,
                color: Colors.grey,
                child: Image.asset(
                  'assets/images/student.jpg',
                  fit: BoxFit.fill,
                ),
              ),

              const Text("BoxFit.fitWidth"),
              Container(
                width: 300,
                height: 150,
                color: Colors.grey,
                child: Image.asset(
                  'assets/images/student.jpg',
